cmake_minimum_required(VERSION 3.16)
project(foton_tests CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(NOT CMAKE_BUILD_TYPE)
  set(CMAKE_BUILD_TYPE Release)
endif()
set(CORE ${CMAKE_CURRENT_SOURCE_DIR}/../core)
add_executable(foton_tests test_main.cpp ${CORE}/image.cpp ${CORE}/burst.cpp ${CORE}/finish.cpp ${CORE}/outputs.cpp)
target_compile_options(foton_tests PRIVATE -O3 -Wall -Wextra)
find_package(Threads REQUIRED)
target_link_libraries(foton_tests Threads::Threads)
